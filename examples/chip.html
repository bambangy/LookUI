JavaScript

const tag = Look.lkChip('#tag', {
  label: 'JavaScript',
  color: 'primary',          // primary | secondary | accent | positive | warning | negative | info
  icon: 'star-fill',         // or avatar: 'JS' | '/img/me.png' | element
  filled: false, outline: false, square: false, dense: false,
  removable: true,
  onRemove(chip) { return confirm('Remove?'); }, // return false to keep it
  onSelect(selected) { console.log(selected); },  // makes it a toggle chip
});

tag.label = 'TypeScript';
tag.selected = true;
tag.enabled = false;
tag.remove();   // animated
tag.destroy();
