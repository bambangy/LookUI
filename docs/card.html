JavaScript — accordion header

const card = Look.lkCard('#details', {
  collapsed: true,
  onOpen(c) { console.log('opened'); },
  onClose(reason, c) { console.log('closed by', reason); },   // 'header' | 'api'
});

card.toggle();        // slide open/closed (fires callbacks with reason 'api')
card.expanded = true; // silent
card.destroy();       // restores the original markup
