JavaScript

const badge = Look.lkPingBadge('#notification-btn', {
  count: 3,                         // null / 0 = dot only, > 99 shows "99+"
  position: 'top-right',            // top-right | top-left | bottom-right | bottom-left
  colorClass: 'lk-badge--negative',
  pulse: true,
});

badge.setCount(12);
badge.hide();
badge.show();
badge.destroy();
