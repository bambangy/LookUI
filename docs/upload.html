Form & validation

JavaScript

const cv = Look.lkUpload('#cv', {
  label: 'Resume',
  mode: 'box',                 // 'field' (default) | 'box'
  accept: '.pdf,.docx',        // same syntax as the native attribute
  maxSize: '5MB',              // per file: bytes or '500KB' / '5MB' / '1.5GB'
  rules: ['required'],         // validation like every form component
  onChange(value, ctx) { console.log(value, ctx.source); },
  onReject(rejected) { console.log(rejected.map((r) => r.message)); },
});

cv.value;        // File | null   (File[] when multiple)
cv.clear();
cv.destroy();
