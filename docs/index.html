Quick start

No build step and no dependencies: one CSS file, one JS file, then call factories on your own markup.

1 · Install

npm install @lookui/core

…or load it from a CDN (pin the version in production):

<link rel="stylesheet" href="https://unpkg.com/@lookui/core@0.3.0/dist/look.min.css">
<script src="https://unpkg.com/@lookui/core@0.3.0/dist/look.min.js"></script>

2 · Include

<!-- classic script: exposes window.Look -->
<link rel="stylesheet" href="node_modules/@lookui/core/dist/look.css">
<script src="node_modules/@lookui/core/dist/look.js"></script>
// bundlers / ES modules
import '@lookui/core/look.css';
import { lkTextbox, lkDropdown, lkValidation, lkToast } from '@lookui/core';

3 · Markup + factories

<form id="signup">
  <input id="email">
  <select id="plan"></select>
  <button class="lk-btn lk-btn--primary">Sign up</button>
</form>

<script>
  const email = Look.lkTextbox('#email', {
    label: 'Email', name: 'email', rules: ['required', 'email'], validate: true,
  });
  const plan = Look.lkDropdown('#plan', {
    label: 'Plan', name: 'plan', required: true,
    items: [{ value: 'free', label: 'Free' }, { value: 'pro', label: 'Pro' }],
  });
  const form = Look.lkValidation(email, plan);

  document.getElementById('signup').addEventListener('submit', (e) => {
    e.preventDefault();
    if (form.validate().hasError) return;
    Look.lkToast({ type: 'positive', title: 'Welcome!', message: email.value });
  });
</script>

4 · Clean up

Every factory returns a plain object. Its owner (your controller / page script) calls destroy() when the markup goes away; composables also have close() / hide().
