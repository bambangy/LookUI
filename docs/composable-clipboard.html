JavaScript

const copy = Look.lkClipboard('#copy-btn', {
  target: '#token',                 // or text: 'literal' | text: () => compute()
  onCopy(text, ctx) { console.log('copied', text, ctx.source); },
  onError(err) { console.warn(err); },
  beforeCopy(text) { return text.trim(); },   // false = cancel, string = replace
});

await copy.copy('something else');  // programmatic, source 'api'
copy.destroy();

await Look.lkClipboard.copy('plain text');   // no trigger needed -> true / false
const pasted = await Look.lkClipboard.read(); // string, or null when denied
