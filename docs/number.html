JavaScript

const qty = Look.lkNumber('#qty', {
  min: 0, max: 99, step: 1,
  spinners: true,            // + / - buttons; ArrowUp/Down, PageUp/Down = 10 steps
  wheel: true,               // mouse wheel while focused (opt-in)
  onChange(value, ctx) { console.log(value, ctx.source, ctx.display); },
});

const price = Look.lkNumber('#price', { format: 'currency', currency: 'IDR', decimals: 0 });
const rate  = Look.lkNumber('#rate',  { format: 'percent', value: 0.15 });   // value is a fraction

qty.value;           // number | null   (setter is silent)
qty.destroy();
